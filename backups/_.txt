function renderByBrand() {
  const groups = {};

  allPhones.forEach(p => {
    if (!groups[p.brand]) groups[p.brand] = [];
    groups[p.brand].push(p);
  });

  renderSections(groups, Object.keys(groups).sort());
}

function renderByType() {
  const groups = {};

  allPhones.forEach(p => {
    const type = p.type || "other";
    if (!groups[type]) groups[type] = [];
    groups[type].push(p);
  });

  renderSections(groups, Object.keys(groups).sort());
}

function renderByYear() {
  const groups = {};

  allPhones.forEach(p => {
    const year = p.year || "unknown";
    if (!groups[year]) groups[year] = [];
    groups[year].push(p);
  });

  const orderedYears = Object.keys(groups)
    .map(y => (y === "unknown" ? y : Number(y)))
    .sort((a, b) => {
      if (a === "unknown") return 1;
      if (b === "unknown") return -1;
      return b - a; // newest first
    });

  renderSections(groups, orderedYears);
}

function createPhoneCard(phone) {
  const card = document.createElement("div");
  card.className = "phone";

  const wikiLink = `https://lpcwiki.miraheze.org/wiki/${encodeURIComponent(phone.brand + " " + phone.model)}`;
  const editLink = wikiLink + "?action=edit";

  const badgeHTML = phone.clone
    ? `<div class="clone-badge">clone</div>`
    : "";

  card.innerHTML = `
    <div class="phone-thumb">
      ${badgeHTML}
      <img src="${phone.image || "./placeholder.png"}" alt="${phone.brand} ${phone.model}">
    </div>

    <div class="phone-info">
      <div class="phone-title">${phone.brand} ${phone.model}</div>
      <div class="phone-meta">${phone.type || "unknown"} · ${phone.year || "unknown"}</div>

      ${
        phone.article
          ? `<a class="watch-btn green" href="${wikiLink}" target="_blank">find out more →</a>`
          : `
            <div class="no-article">no wiki article yet</div>
            <a class="watch-btn green" href="${editLink}" target="_blank">want to create it?</a>
          `
      }
    </div>
  `;

  return card;
}
